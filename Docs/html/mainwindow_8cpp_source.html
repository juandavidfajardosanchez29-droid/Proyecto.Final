251void MainWindow::exportarCSV()
252{
253 QString ruta = QFileDialog::getSaveFileName(
254 this, "Guardar reporte", "inventario.csv", "Archivos CSV (*.csv)");
255 if (ruta.isEmpty())
256 return;
257
258 // 1. Leer la base de datos y crear objetos Articulo
259 QSqlQuery q("SELECT a.id, a.nombre, a.categoria_id, a.cantidad, "
260 "a.stock_minimo, a.ubicacion, a.fecha_adquisicion, c.nombre "
261 "FROM articulos a JOIN categorias c ON a.categoria_id = c.id "
262 "ORDER BY a.nombre");
263
264 QVector<Articulo> articulos;
265 QStringList categorias;
266 while (q.next()) {
267 articulos.append(Articulo(q.value(0).toInt(), q.value(1).toString(),
268 q.value(2).toInt(), q.value(3).toInt(),
269 q.value(4).toInt(), q.value(5).toString(),
270 q.value(6).toString()));
271 categorias.append(q.value(7).toString());
272 }
273
274
275 QFile archivo(ruta);
276 if (!archivo.open(QIODevice::WriteOnly | QIODevice::Text)) {
277 QMessageBox::warning(this, "Error", "No se pudo crear el archivo.");
278 return;
279 }
280
281 QTextStream out(&archivo);
282 out.setEncoding(QStringConverter::Utf8);
283 out.setGenerateByteOrderMark(true); // para que Excel lea bien las tildes
284
285 // Pone el texto entre comillas y duplica las comillas internas
286 auto celda = [](const QString &texto) {
287 QString t = texto;
288 t.replace("\"", "\"\"");
289 return "\"" + t + "\"";
290 };
291
292 // 3. Escribir encabezado y filas
293 out << "Nombre;Categoria;Cantidad;Stock minimo;Ubicacion;Fecha;Estado\n";
294 for (int i = 0; i < articulos.size(); ++i) {
295 const Articulo &a = articulos.at(i);
296 out << celda(a.nombre()) << ";"
297 << celda(categorias.at(i)) << ";"
298 << a.cantidad() << ";"
299 << a.stockMinimo() << ";"
300 << celda(a.ubicacion()) << ";"
301 << celda(a.fecha()) << ";"
302 << (a.stockBajo() ? "STOCK BAJO" : "OK") << "\n";
303 }
304
305 archivo.close();
306 QMessageBox::information(this, "Reporte",
307 "Se exportaron " + QString::number(articulos.size())
308 + " artículos.");
309}
